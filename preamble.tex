% \entryheader{type}{YYYY-MM-DD}{title}
% type is one of: research, analysis, meeting, design, reflection
\newcommand{\entryheader}[3]{%
  \section[#2: #3]{#3}%
  \begin{tcolorbox}[colback=#1!8,colframe=#1,boxrule=0.6pt,arc=2pt,
      left=6pt,right=6pt,top=3pt,bottom=3pt]
    \textbf{\MakeUppercase{#1}}\hfill\textbf{#2}
  \end{tcolorbox}%
}

% Optional tags line: \entrytags{synthetic-eeg, aad}
\newcommand{\entrytags}[1]{\noindent\textit{Tags:} #1\par}

% Labelled field, used for meeting minutes and entry sections
\newcommand{\entryfield}[1]{\par\smallskip\noindent\textbf{#1}\par\nopagebreak}

% Meeting minutes: \begin{meeting}{YYYY-MM-DD}{Title}{team/supervisor/class session}
\newenvironment{meeting}[3]{%
  \entryheader{meeting}{#1}{#2}%
  \noindent\textit{Type:} #3\par
}{\par}

% Entry: \begin{entry}{type}{YYYY-MM-DD}{Title}
\newenvironment{entry}[3]{%
  \entryheader{#1}{#2}{#3}%
}{\par}
